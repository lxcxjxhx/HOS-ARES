# DeepAudit FastAPI 后端 AI 审计平台（服务式，uvicorn app.main:app）。
# 本清单基于源码实际 import 归纳 (python3/pip 自带标准库不计)。
# 依据:
#   - app/main.py                       -> fastapi
#   - run.sh serve                      -> uvicorn
#   - app/db/session.py                 -> sqlalchemy (async); 默认 DATABASE_URL 为 postgresql+asyncpg
#   - app/core/config.py 等             -> pydantic, pydantic_settings
#   - app/services/scanner.py 等        -> httpx
#   - app/core/encryption.py 等         -> cryptography
#   - app/core/security.py              -> passlib
#   - app/services/report_generator.py  -> jinja2, weasyprint
#   - app/schemas/user.py 等            -> EmailStr (需 email-validator)
#   - app/api/v1/endpoints/*.py         -> UploadFile/Form (需 python-multipart)
# 注: weasyprint 运行时需要系统库(pango/cairo)，如缺少需另行 apk add。
fastapi
uvicorn
sqlalchemy
asyncpg
pydantic
pydantic-settings
httpx
cryptography
passlib
bcrypt
jinja2
weasyprint
email-validator
python-multipart
