# 03 · 数据模型与状态机

> 目标：任何"卡住/重复/丢结果"的问题，都能在表结构和状态机上找到答案。
>
> **v3 变更**：新增 `routesets` 与 `route_bindings` 两张表（`27`）。
> `task_events` 的 `kind` 增加 `route` / `rediscovery`。
> `capsules` 增加 `routeset_sha256` 与 `route_ref`（可空，兼容 v1.0 capsule）。

## 1. 表结构（SQLite 起步，DDL 与 PG 兼容）

```sql
-- 任务：一切调度的中心
CREATE TABLE tasks (
  id            TEXT PRIMARY KEY,           -- ulid
  engagement_id TEXT NOT NULL,
  idem_key      TEXT NOT NULL,              -- sha256(engagement|asset|oracle|params)
  profile_name  TEXT NOT NULL,
  profile_hash  TEXT NOT NULL,
  asset_id      TEXT NOT NULL,
  oracle_ids    TEXT NOT NULL,              -- json array
  state         TEXT NOT NULL DEFAULT 'QUEUED',
  lease_owner   TEXT,                       -- worker_id
  lease_until   INTEGER,                    -- unix 秒（Redis 授时）
  heartbeat_at  INTEGER,
  attempt       INTEGER NOT NULL DEFAULT 0,
  max_attempts  INTEGER NOT NULL DEFAULT 3,
  error_class   TEXT,                       -- retryable|terminal|needs_human
  error_code    TEXT,
  budget_json   TEXT NOT NULL,              -- 已消耗 steps/tokens/wall/cost/rediscovery
  routeset_sha256 TEXT,                     -- ★ 本次任务使用的绑定集
  result_json   TEXT,
  created_at    INTEGER NOT NULL,
  updated_at    INTEGER NOT NULL
);
CREATE UNIQUE INDEX ux_tasks_idem ON tasks(idem_key);
CREATE INDEX ix_tasks_pick   ON tasks(state, lease_until);
CREATE INDEX ix_tasks_engage ON tasks(engagement_id, state);

-- ★ 状态黑板：只存事实，不存判断/地址/凭据（30 §2.1，不变量 I10：只增不改）
CREATE TABLE blackboard_facts (
  id            TEXT PRIMARY KEY,           -- ulid
  task_id       TEXT NOT NULL,
  subject       TEXT NOT NULL,              -- 例 "web-01"
  predicate     TEXT NOT NULL,              -- 例 "server"
  value         TEXT NOT NULL,              -- 例 "nginx/1.24"
  origin        TEXT NOT NULL,              -- observed|target|tool|inference
  from_target   INTEGER NOT NULL DEFAULT 0, -- ★ 1 = 来自目标响应（不可信，禁止当指令）
  evidence_id   TEXT,                       -- 出处（capsule/事件 id），必填
  supersedes    TEXT,                       -- 取代哪条旧事实（旧的不删）
  ttl_s         INTEGER,
  created_at    INTEGER NOT NULL
);
CREATE INDEX ix_facts_task ON blackboard_facts(task_id, created_at);
CREATE INDEX ix_facts_sp   ON blackboard_facts(task_id, subject, predicate);
-- 冲突（同 subject+predicate 不同 value）不合并：两条都留，查询时标 disputed

-- ★ 绑定集：内容寻址，一次 discovery 一行
CREATE TABLE routesets (
  sha256        TEXT PRIMARY KEY,           -- = routeset_sha256
  asset_id      TEXT NOT NULL,
  profile_hash  TEXT NOT NULL,
  resolver_version TEXT NOT NULL,
  query_hash    TEXT NOT NULL,              -- sha256(canonical_json(RouteQuery))
  sources_used  TEXT NOT NULL,              -- json array
  binding_count INTEGER NOT NULL,
  path          TEXT NOT NULL,              -- out/routes/<sha>.json（含模板）
  created_at    INTEGER NOT NULL,
  expires_at    INTEGER NOT NULL            -- TTL 汇总（最早过期时间）
);
CREATE INDEX ix_routeset_asset ON routesets(asset_id, created_at);

-- ★ 绑定明细：便于按 ref / 标签 / 来源查询，不必解 JSON
CREATE TABLE route_bindings (
  routeset_sha256 TEXT NOT NULL,
  ref             TEXT NOT NULL,            -- route://<source>/<opaque16>
  source          TEXT NOT NULL,            -- contract|observed|crawl|probe|operator
  sources_all     TEXT NOT NULL,            -- json array（多来源佐证）
  confidence      REAL NOT NULL,
  template_sha256 TEXT NOT NULL,            -- 不含模板正文（正文只在文件里）
  semantic_tags   TEXT NOT NULL,            -- json array
  auth_requirement TEXT,
  ttl_s           INTEGER NOT NULL,
  PRIMARY KEY (routeset_sha256, ref)
);
CREATE INDEX ix_binding_ref ON route_bindings(ref);

-- 事件流：step.jsonl 的库内镜像（可观测性主表）
CREATE TABLE task_events (
  seq        INTEGER PRIMARY KEY AUTOINCREMENT,
  task_id    TEXT NOT NULL,
  ts         INTEGER NOT NULL,
  -- kind 增加 route / rediscovery
  kind       TEXT NOT NULL,   -- plan|binding|preflight|http|oracle|judge|capsule|budget
                              -- |scope|throttle|approval|error|route|rediscovery
  oracle_id  TEXT,
  tool       TEXT,
  route_ref  TEXT,            -- ★ 这步打的是哪条绑定
  elapsed_ms INTEGER,
  tokens     INTEGER DEFAULT 0,
  ok         INTEGER NOT NULL,
  err_class  TEXT,
  detail_json TEXT
);
CREATE INDEX ix_events_task ON task_events(task_id, seq);
CREATE INDEX ix_events_route ON task_events(route_ref);

-- 发现：只引用证据，不复制证据
CREATE TABLE findings (
  id            TEXT PRIMARY KEY,
  task_id       TEXT NOT NULL,
  asset_id      TEXT NOT NULL,
  oracle_id     TEXT NOT NULL,
  oracle_version TEXT NOT NULL,
  title         TEXT NOT NULL,
  severity      TEXT NOT NULL,              -- info|low|medium|high|critical
  confidence    REAL NOT NULL,              -- 规则算出，非模型
  route_strength TEXT,                      -- ★ observed|contract|crawl|probe
  cwe           TEXT,
  state         TEXT NOT NULL,              -- CANDIDATE|VERIFIED|REFUTED|UNVERIFIABLE|BLOCKED
  capsule_id    TEXT,                       -- NULL 且 state=VERIFIED 视为不合法
  created_at    INTEGER NOT NULL
);
CREATE UNIQUE INDEX ux_finding_run ON findings(task_id, oracle_id, asset_id);

-- 证据包：内容寻址，sealed 后不可变
CREATE TABLE capsules (
  id           TEXT PRIMARY KEY,            -- = self_sha256
  finding_id   TEXT NOT NULL,
  path         TEXT NOT NULL,               -- out/capsules/<id>.json
  self_sha256  TEXT NOT NULL,
  prev_sha256  TEXT,                        -- 哈希链
  routeset_sha256 TEXT,                     -- ★ NULL 仅允许 capsule_version=1.0
  route_ref    TEXT,                        -- ★ 主绑定的 ref
  sealed_at    INTEGER NOT NULL,
  replay_status TEXT DEFAULT 'NEVER',       -- NEVER|OK|PARTIAL|FAILED|BLOCKED
                                            -- |ROUTE_UNRESOLVED|ROUTE_DRIFT
  replay_at    INTEGER
);

-- 审批门
CREATE TABLE approvals (
  id           TEXT PRIMARY KEY,
  task_id      TEXT NOT NULL,
  action_class TEXT NOT NULL,               -- reversible_write|scope_expand|discovery_probe
  impact_json  TEXT NOT NULL,
  state        TEXT NOT NULL DEFAULT 'PENDING', -- PENDING|APPROVED|REJECTED|EXPIRED
  requested_at INTEGER NOT NULL,
  decided_at   INTEGER,
  decided_by   TEXT,
  reason       TEXT
);

-- 跑分
CREATE TABLE bench_runs (
  id            TEXT PRIMARY KEY,
  profile_name  TEXT NOT NULL,
  started_at    INTEGER NOT NULL,
  cases_total   INTEGER NOT NULL,
  tp INTEGER, fp INTEGER, fn INTEGER, tn INTEGER,
  precision     REAL, recall REAL, f1 REAL,
  replay_pass   INTEGER,
  first_finding_steps INTEGER,
  routeset_sha256 TEXT,                      -- ★ N6：三轮必须相同
  tokens        INTEGER,
  cost_usd      REAL,
  wall_s        REAL,
  voided        INTEGER DEFAULT 0,           -- 越界即 1，该次跑分作废
  notes         TEXT
);
```

## 2. 任务状态机（8 态，闭环）

```
                     POST /tasks（idem_key 去重）
                              │
                              ▼
                          ┌───────┐   取消(人工)   ┌───────────┐
                          │QUEUED │───────────────▶│ CANCELLED │
                          └───┬───┘                └───────────┘
                     领租约 │ ▲ 租约过期回收(attempt+1)
                              ▼ │
                          ┌───────┐
                          │LEASED │
                          └───┬───┘  开始执行
                              ▼
                          ┌───────┐   遇高危动作   ┌──────────────────┐
                          │RUNNING│◀──────────────▶│AWAITING_APPROVAL │
                          └───┬───┘   审批通过/拒绝  └──────────────────┘
                    ┌─────────┴─────────┐
              成功  ▼                   ▼ 失败
                ┌──────┐          ┌────────┐
                │ DONE │          │ FAILED │
                └──────┘          └───┬────┘
                                      │ attempt ≥ max_attempts 或 terminal
                                      ▼
                                ┌─────────────┐
                                │ DEAD_LETTER │
                                └─────────────┘
```

**迁移唯一入口**：`store.transition(task_id, from_state, to_state, **fields)`，带 `WHERE state=from` 条件更新，返回影响行数；0 行说明被抢，调用方必须放弃当前工作（防双跑）。

```python
def transition(task_id, frm, to, **fields) -> bool:
    n = db.execute("UPDATE tasks SET state=?, updated_at=?, "+set_clause+
                   " WHERE id=? AND state=?",
                   [to, now(), *fields.values(), task_id, frm]).rowcount
    if n == 0:
        log.warn("transition_lost", task_id=task_id, frm=frm, to=to)  # 被别人抢了
    return n == 1
```

**没有这一步，"重复执行"必然发生。** 这是幂等之外的第二道闸。

**v3 补充**：`RUNNING` 状态**不因重发现而迁移**。
`rediscovery` 是任务内部的子动作（记 `task_events.kind='rediscovery'`），
不改变任务状态，但它消耗 `budget.rediscovery` 配额。

## 3. 租约规则（数字先定死）

| 参数 | 值 | 说明 |
|---|---|---|
| 心跳间隔 | 5 s | worker 每 5 s 续租并写 `heartbeat_at` |
| 租约 TTL | 30 s | `lease_until = now + 30` |
| 回收扫描 | 5 s | `SELECT id FROM tasks WHERE state IN ('LEASED','RUNNING') AND lease_until < now()` |
| 重排规则 | `attempt+1`；`attempt ≥ 3` → `DEAD_LETTER` | 重排保留已产出的 capsule，不重跑已验证 oracle |
| 心跳超时但进程还活着 | 立即自杀式退出（`os._exit`） | 避免双跑；日志写明 `lease_lost_suicide` |

**为什么是租约不是分布式锁**：锁要释放、会残留；租约自带过期，worker 崩了无需他人善后。

## 4. 发现状态机

```
CANDIDATE ──judge=VERIFIED──▶ VERIFIED       （必须有 capsule，否则降级）
    ├─────judge=REFUTED────▶ REFUTED         （进负样本库，可回灌用例）
    ├─────judge=INCONCLUSIVE▶ UNVERIFIABLE   （报告"未能验证"单列）
    └─────guard 拒绝───────▶ BLOCKED         （越界/未授权/超额，永不进报告正文）
```

**硬约束**：`state='VERIFIED' AND capsule_id IS NULL` 在 04 的日检里视为数据损坏，报告渲染直接拒绝。

## 5. capsule 状态

```
DRAFT ──seal()──▶ SEALED ──replay()──▶ SEALED(replay_status =
                    │                    OK | PARTIAL | FAILED | BLOCKED
                    │                  | ROUTE_UNRESOLVED | ROUTE_DRIFT)
                    ├─ profile_hash 变更 ─▶ STALE（仍可读，replay 时提示需重跑）
                    └─ 绑定过期且无法重解析 ─▶ SEALED(ROUTE_UNRESOLVED 标记)
```

SEALED 后禁止字段修改（`evidence.py` 只暴露不可变 dataclass）。要改就新建 capsule，旧的保留 —— 审计要求可追溯而不是可覆盖。

**v3 补充**：`STALE` 与 `ROUTE_UNRESOLVED` 是两种不同的失效，不能合并：

| 失效类型 | 原因 | 报告动作 |
|---|---|---|
| `STALE`（profile_hash 变） | 配置改了 | "需按新配置重跑" |
| `ROUTE_UNRESOLVED` | 面在当前环境不存在了 | "该测试面已不存在"（**不等于已修复**，见 `06 §5.2`） |
| `ROUTE_DRIFT` | 面还在但结构变了 | "需人工确认后重判" |

## 6. 索引与容量预算（避免"跑两天就慢"）

| 表 | 30 天预估行数 | 处理 |
|---|---|---|
| task_events | ~50 万（120 步 × 4000 任务） | 按月分区；`detail_json` 只存 <=2KB；完整 trace 落 `out/` 文件 |
| tasks | ~1 万 | 保持热表 |
| capsules | ~1 万 | 文件在盘上，库只存索引与哈希 |
| **blackboard_facts** | ~数千（每任务几十到几百条事实） | 随任务保留；被 capsule 引用的永久留 |
| **routesets** | 数千（每资产每任务一次，命中缓存后大幅下降） | 只存元数据；模板在 `out/routes/`；按 `expires_at` 定期清理过期行（文件保留 30 天供审计） |
| **route_bindings** | 数万（每 routeset 数条到数百条） | 随 routeset 级联清理 |
| bench_runs | 数百 | 永久保留（对外可信度） |

**归档策略**：`task_events` 90 天后导出 parquet 并清表；capsule 文件永不删（WORM 语义，见 06）。
`routesets` 与 `route_bindings` 属**过程数据**，30 天后可清理（但要保留被 capsule 引用的那些）。

## 7. 一致性自检（`make doctor --deep` 包含）

```sql
-- 1 重复幂等键（应为 0）
SELECT idem_key, COUNT(*) c FROM tasks GROUP BY idem_key HAVING c>1;
-- 2 卡死任务（LEASED/RUNNING 且租约过期 >5 分钟，应为 0）
SELECT id FROM tasks WHERE state IN('LEASED','RUNNING') AND lease_until < strftime('%s','now')-300;
-- 3 无证据的已验证发现（应为 0）
SELECT id FROM findings WHERE state='VERIFIED' AND capsule_id IS NULL;
-- 4 孤儿 capsule（应为 0）
SELECT c.id FROM capsules c LEFT JOIN findings f ON f.id=c.finding_id WHERE f.id IS NULL;
-- 5 越界事件（必须 0，出现即事故）
SELECT COUNT(*) FROM task_events WHERE kind='scope' AND ok=0 AND detail_json LIKE '%VIOLATION%';

-- ── v3 新增五条 ─────────────────────────────────────────────
-- 6 VERIFIED 的发现缺 route 证据（应为 0）
SELECT id FROM capsules
 WHERE capsule_version <> '1.0' AND (routeset_sha256 IS NULL OR route_ref IS NULL);
-- 7 引用了不存在的绑定集（应为 0）
SELECT c.id FROM capsules c
 LEFT JOIN routesets r ON r.sha256 = c.routeset_sha256
 WHERE c.routeset_sha256 IS NOT NULL AND r.sha256 IS NULL;
-- 8 弱绑定给了高严重度（应为 0；这是措辞纪律的数据侧保证）
SELECT id FROM findings
 WHERE route_strength IN ('crawl','probe') AND severity IN ('high','critical');
-- 9 重发现超配额的任务（应为 0）
SELECT id FROM tasks WHERE budget_json LIKE '%"rediscovery":%"' AND
  CAST(json_extract(budget_json,'$.rediscovery.used') AS INTEGER) >
  CAST(json_extract(budget_json,'$.rediscovery.max') AS INTEGER);
-- 10 绑定过期但任务仍在跑（>5 分钟，应为 0；说明 TTL 与续租脱节）
SELECT t.id FROM tasks t JOIN routesets r ON r.sha256 = t.routeset_sha256
 WHERE t.state='RUNNING' AND r.expires_at < strftime('%s','now')-300;
-- 11 ★ 来源越界（有人绕过 resolver 用了不在 RouteSet 里的 ref；必须 0）
SELECT seq, task_id, detail_json FROM task_events
 WHERE kind='binding' AND ok=0 AND detail_json LIKE '%UNBOUND%';
-- 12 ★ 事实无出处（黑板每条事实必须有 evidence_id；必须 0）
SELECT id FROM blackboard_facts WHERE evidence_id IS NULL OR evidence_id='';
-- 13 ★ 目标数据被当成事实却未标记（from_target=0 但 origin='target'；必须 0）
SELECT id FROM blackboard_facts WHERE origin='target' AND from_target=0;
```

第 5/11/12/13 条是**上线门禁**：不为 0 不许交付。
第 6–10 条是 v3 的**路由卫生门禁**，第 11 条是**来源越界门禁**，第 12–13 条是**黑板可信度门禁**：它们让"路由层悄悄腐化"变成可查的数据问题，
而不是等到某天发现代码里又出现了 `if not found: path = ...`。
